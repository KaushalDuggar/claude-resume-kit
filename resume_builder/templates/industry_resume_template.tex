%

% Macros
% Entry header: {Org}{Date}{Role}{Location}. Date is ALWAYS top-right.
% No outer bullet: company/date row sits flush left, sub-bullets indent under it.
\newcommand{\resumeSubheading}[4]{%
  \item
  \begin{tabular*}{\linewidth}[t]{@{}l@{\extracolsep{\fill}}r@{}}
    \textbf{#1} & #2 \\
    \textit{\small #3} & \textit{\small #4} \\
  \end{tabular*}%
}
% Bullet: {Bold label}{Text}. Keep each to exactly 2 rendered lines (~185 chars incl. label).
\newcommand{\resumeItem}[2]{\item \small{\textbf{#1:} #2}}

\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0in, label={}, itemsep=5pt, topsep=3pt, parsep=0pt, partopsep=0pt]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{\begin{itemize}[leftmargin=0.18in, label=$\circ$, itemsep=1pt, topsep=1pt, parsep=0pt, partopsep=0pt]}
\newcommand{\resumeItemListEnd}{\end{itemize}}
